set(OI_DB_SOURCES src/oi_db.cpp)
if(OI_ENABLE_MYSQL)
	find_path(OI_MYSQL_INCLUDE_DIR NAMES mysql.h PATH_SUFFIXES mysql mariadb)
	find_library(OI_MYSQL_LIBRARY NAMES mysqlclient libmysql mariadb libmariadb)
	if(NOT OI_MYSQL_INCLUDE_DIR OR NOT OI_MYSQL_LIBRARY)
		message(FATAL_ERROR "OI_ENABLE_MYSQL requires MySQL or MariaDB client headers and library")
	endif()
	list(APPEND OI_DB_SOURCES src/oi_mysql.cpp)
endif()
add_library(db_static STATIC ${OI_DB_SOURCES})
add_library(db_m_static STATIC ${OI_DB_SOURCES})
foreach(oi_db_target db_static db_m_static)
	target_include_directories(${oi_db_target} PUBLIC inc)
	if(OI_ENABLE_MYSQL)
		target_include_directories(${oi_db_target} SYSTEM PRIVATE "${OI_MYSQL_INCLUDE_DIR}")
		target_link_libraries(${oi_db_target} PUBLIC "${OI_MYSQL_LIBRARY}")
		target_compile_definitions(${oi_db_target} PUBLIC OI_ENABLE_MYSQL=1)
	endif()
endforeach()
target_compile_definitions(db_static PUBLIC _SINGLE_THREAD_)
